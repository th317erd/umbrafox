userland.on("navigation", event => {
  if (event.href.startsWith("zoom://")) {
    event.preventDefault();
    return;
  }

  const url = new URL(event.href);
  if (url.pathname == "/old") {
    event.href = new URL("/new", location.href).href;
  }
});
